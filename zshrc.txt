    for symbol in "$@"
    do
      rem $symbol
    done

  watchlist=$(grep "^US=" ~/workspaces/stocks/.env)
  unset watchlistNew
  for symbol in $(echo ${tickers[@]});
  do
    echo $watchlist | grep "\b${symbol}\b" > /dev/null
    if [ ! $? -eq 0 ]; then
      echo "\n$symbol added"
      watchlistNew+=" $symbol"
    fi
  done

  # prepend to existing list
  watchlist=$(echo $watchlist | sed "s/US=//")
  watchlistNew+=" $watchlist"

  echo "Entry?"
  read numerator

  echo "Stop level"
  read stop
  if [[ -z $stop ]]; then
    echo "Stop Percetange"
    read stopPerc
    stop=$(bc -l <<< "scale=2; $numerator*(1-$stopPerc/100)")
  fi
  if [[ -z $stop ]]; then
    stop=$(echo "scale=2; (0.96 * $numerator)" | bc )
    echo Stop: $stop
  fi

  echo "Target"
  read target

  multiplier=1
  if [[ ! -z $2 ]]; then
    multiplier=$2
  fi

  equity=$(( multiplier * $1 ))
  rr=$3
  if [[ -z $rr ]]; then
    rr=2
  fi

  # numerator=$(echo $1 | awk '{split($1, arr, "/"); print arr[1]}')
  denominator=$stop
  target=$target
  target2=$target
  result=$(echo "scale=2; $equity / ($numerator - $denominator)" | bc)
  total=$(echo "scale=2; $numerator * $result" | bc )
  targetPerc=$(echo "scale=2; ($target-$numerator)/$numerator" | bc)

  if [[  -z $target ]]; then
    echo "R multiple $rr"
    target=$(echo "scale=2; ($numerator) + (($numerator - $denominator) * 3)" | bc )
    target2=$(echo "scale=2; ($numerator) + (($numerator - $denominator) * $rr)" | bc )
  fi

  if [[ $targetPerc -gt 0.2 ]]; then
    echo "\n${Red}target is > 20%$NC"
    target=$(echo "scale=2; ($numerator)*1.2" | bc )
  fi


  echo "${Green}$result${NC} $numerator/$denominator/$target/$target2 ${Red}$total${NC}"
  echo "save $result $numerator/$denominator/$target/$target2" | sed "s/\n//g" | pbcopy

  echo "\nWant to revise?"
  read yesno
  case $yesno in
      [Yy]* )
          echo "Enter size"
          read multiplier
          equity=$(( multiplier * $1 ))

          result=$(echo "scale=2; $equity / ($numerator - $denominator)" | bc)
          total=$(echo "scale=2; $numerator * $result" | bc )
          echo "${Green}$result${NC} $numerator/$denominator/$target/$target2 ${Red}$total${NC}"
          echo "save $result $numerator/$denominator/$target2/$target" | sed "s/\n//g" | pbcopy
      ;;
      [Nn]* )
          psWork $1 $2
      ;;
  esac
  echo "=============\n\n"
  psWork $1 $2
}
sp() {
  if [[ $# -lt 2 ]]; then
    echo "need both price & equity"
    exit1
  fi

  spWork $@
}

spWork() {
  multiplier=1
  if [[ ! -z $2 ]]; then
    multiplier=$2
  fi

  equity=$(( multiplier * $1 ))

  rr=$3
  if [[ -z $rr ]]; then
    rr=2
  fi

  echo "Entry"
  read numerator

  echo "Stop level"
  read stop
  if [[ -z $stop ]]; then
    stop=$(echo "scale=2; (1.04 * $numerator)" | bc )
    echo Stop: $stop
  fi

  echo "Target"
  read target

  denominator=$stop
  result=$(echo "scale=2; $equity / ($denominator - $numerator)" | bc)
  total=$(echo "scale=2; $denominator * $result" | bc )
  target2=$target

  if [[  -z $target ]]; then
    echo "R multiple $rr\n"
    target=$(echo "scale=2; $numerator - (($denominator - $numerator) * 3)" | bc )
    target2=$(echo "scale=2; $numerator - (($denominator - $numerator) * $rr)" | bc )
  fi

  targetPerc=$(echo "scale=2; ($numerator-$target)/$numerator" | bc)
  if [[ $targetPerc -gt 0.2 ]]; then
    echo "\n${Red}target is > 20%$NC"
    target=$(echo "scale=2; ($numerator)*0.8" | bc )
  fi


  echo "${Green}$result${NC} $numerator/$denominator/$target/$target2 -${Red}$total${NC}"
  echo "evas $result $numerator/$denominator/$target/$target2" | pbcopy

  echo "\nWant to revise?"
  read yesno
  case $yesno in
      [Yy]* )
          echo "Enter size"
          read multiplier
          equity=$(( multiplier * $1 ))

          result=$(echo "scale=2; $equity / ($denominator - $numerator)" | bc)
          total=$(echo "scale=2; $denominator * $result" | bc )
          echo "${Green}$result${NC} $numerator/$denominator/$target/$target2 -${Red}$total${NC}"
          echo "evas $result $numerator/$denominator/$target/$target2" | pbcopy
      ;;
      [Nn]* )
        spWork $1 $2
      ;;
  esac
  spWork $1 $2
}

comma() {
  local input result
  if [ "$#" -gt 0 ]; then
    input="$*"          # arguments: comma ALTS BK CSGS
  else
    input="$(cat)"      # piped: awk ... | comma
  fi
  result=$(printf '%s' "$input" | tr '\n' ' ' | tr -s ' ' | sed 's/^ *//; s/ *$//; s/ /,/g')
  printf '%s\n' "$result"      # show it
  printf '%s' "$result" | pbcopy   # copy it (no trailing newline)
}

srep() {
  word_count=$(echo $@ | tr "," " " | wc -w)
  if [[ $word_count -eq 1 ]]; then
    rank $1 > rank.json
    ./sales.sh > salesreport.txt

    debt_to_equity=$(cat rank.json | jq -r '.content.fundamentalDataBlock.miscBlock.debt')

    cat salesreport.txt

    rm growthrates.txt 2> /dev/null
    echo "\n"
    ./quarters.sh < salesreport.txt | jq ".quarters[]" | while read q; do
        q=$(echo $q | sed "s/\"//g");
        grep $q salesreport.txt | tee /dev/tty >> growthrates.txt;
      done
    cat growthrates.txt | ./growth.sh | tee /dev/tty | jq "select(.sales_growth_percent >= 200 or .current_sales_growth_percent >= 200)"

    eps_yr_est=$(cat datalist.json | jq -r ".stock_list[] | select(.symbol == \"${1}\") | ." | jq ".curr_yr_eps_est_percent_chg")

    ./cagr.sh < salesreport.txt | jq --arg sym "$1" --arg debt "$debt_to_equity" --arg eps_yr_est "$eps_yr_est" '{symbol: $sym, debt_to_equity: $debt, curr_yr_eps_est_percent_chg: $eps_yr_est} + .' | jq "."

    ./cagr.sh < salesreport.txt | jq --arg sym "$1" --arg debt "$debt_to_equity" --arg eps_yr_est "$eps_yr_est" '{symbol: $sym, debt_to_equity: $debt, curr_yr_eps_est_percent_chg: $eps_yr_est} + .' | jq "{eps_consecutive_Y_growth_est, sales_consecutive_Y_growth_est, curr_yr_eps_est_percent_chg}" | pbcopy

    return
  fi

  rm srep.txt 2> /dev/null
  rm srep.json 2> /dev/null
  tickers=$(uncomma $@ | sed "s/\"//g")
  for symbol in $(echo $tickers[@]); do
    echo $symbol
    rank $symbol > rank.json
    ./sales.sh > salesreport.txt
    debt_to_equity=$(cat rank.json | jq -r '.content.fundamentalDataBlock.miscBlock.debt')

    rm growthrates.txt 2> /dev/null
    echo "\n"
    ./quarters.sh < salesreport.txt | jq ".quarters[]" | while read q; do
        q=$(echo $q | sed "s/\"//g");
        grep $q salesreport.txt | tee /dev/tty >> growthrates.txt;
      done

    # echo "\n$symbol =====" >> srep.txt
    cat growthrates.txt | ./growth.sh | jq --arg sym "$symbol" --arg debt "$debt_to_equity" '{symbol: $sym, debt_to_equity: $debt} + .' >> srep.json
    printf "," >> srep.json
    ./cagr.sh < salesreport.txt | jq --arg sym "$symbol" --arg debt "$debt_to_equity" '{symbol: $sym, debt_to_equity: $debt} + .' >> srep.json
    printf "," >> srep.json
  done

  cat srep.json | jq "select(.sales_growth_percent >= 200 or .current_sales_growth_percent >= 200)"
}

toDate() {
  json_date="$1"
  if [[ -z $1 ]]; then
    echo "Enter MS json date"
    read json_date
  fi
  # Extract timestamp using sed
  timestamp=$(echo "$json_date" | sed "s/Date(\(.*\)-.*)/\1/")

  # Convert timestamp from milliseconds to seconds
  timestamp_seconds=$((timestamp / 1000))

  # Convert to readable format
  readable_date=$(date -u -r "$timestamp_seconds" +"%Y, %d %b")

  echo "$Red $readable_date$NC"
}

msdate() {
  if [[ -z $1 ]]; then
    timestamp=$(date +%s)
  else
    timestamp=$(date -j -f "%d-%m-%Y %H:%M:%S UTC" "$1 16:00:00 UTC" +%s)
  fi
  timestamp_ms=$((timestamp * 1000))
  timezone_offset=$(date +%z)
  ms_json_date="/Date($Red${timestamp_ms}$NC${timezone_offset})/"

  echo "$ms_json_date"
}

ddte() {
  #!/bin/bash

# Get today's date in M/D/YYYY format without leading zeros
today=$(date +%m/%d/%Y)

# Convert to YYYY-MM-DD for date calculations
formatted_date=$(echo $today | awk -F'/' '{printf "%d-%d-%d", $3, $1, $2}')

# Get the day of the week (1 = Monday, ..., 6 = Saturday, 7 = Sunday)
day_of_week=$(date -j -f "%Y-%m-%d" "$formatted_date" +%u)

# Check if the day is a weekend
if [ $day_of_week -eq 6 ]; then
    # Saturday: Add 2 days to get to Monday
    output_date=$(date -j -f "%Y-%m-%d" -v+2d "$formatted_date" +%m/%d/%Y)
elif [ $day_of_week -eq 7 ]; then
    # Sunday: Add 1 day to get to Monday
    output_date=$(date -j -f "%Y-%m-%d" -v+1d "$formatted_date" +%m/%d/%Y)
else
    # Weekday: Keep the original date
    output_date=$today
fi

echo $output_date
}

filters() {
  if [[ -n $1 && -f $1 ]]; then
    tickers=$(awk '$3 < 0 {print $2}' "$1" | sort -u)
  else
    echo "Enter positions (Ctrl+D when done):" && myPositions=$(cat)
    tickers=$(echo "$myPositions" | awk '$3 < 0 {print $2}' | sort -u)
  fi
  regex=$(echo "=REGEXMATCH(I:I, \"^(")
  for symbol in $(echo $tickers[@]); do
    regex=$(echo "$regex$symbol|")
  done
  regex=$(echo "$regex)$\")")

  echo $regex | tee /dev/tty | pbcopy
  open "https://docs.google.com/spreadsheets/d/1MlIYs0uha3BrzskcHfchbkLhpFIsNt006YFRl8U4b98/edit?gid=314181279#gid=314181279&fvid=1166312445"

goodReads() {
  if [[ $1 == "list" ]]; then
    file="links.txt"

  if [ ! -t 0 ]; then
      echo "Needs an interactive terminal."
      return
  fi

  total=$(grep -c 'https\?://' "$file")
  count=0

  while IFS= read -r line || [ -n "$line" ]; do
    # skip blank lines and comments
    [[ -z "${line//[[:space:]]/}" ]] && continue
    [[ "$line" == \#* ]] && continue

    url=$(printf '%s\n' "$line" | grep -o 'https\?://[^[:space:]]*' | head -1)
    [ -z "$url" ] && continue

    # title = everything before the url, minus trailing ":" and spaces
    title="${line%%"$url"*}"
    title=$(printf '%s\n' "$title" | sed -e 's/[[:space:]]*$//' -e 's/[:-]$//' -e 's/[[:space:]]*$//')
    [ -z "$title" ] && title="$url"

    if [ -t 1 ]; then
      GREEN=$'\033[1;32m'
      BLUE=$'\033[1;34m'
      RESET=$'\033[0m'
    else
        GREEN=""; BLUE=""; RESET=""
    fi

    count=$((count + 1))
    printf '\n[%d/%d] %s%s%s\n' "$count" "$total" "$GREEN" "$title" "$RESET"
    printf '        %s%s%s\n' "$BLUE" "$url" "$RESET"
    printf 'Read it? [y/N/q] '
    read -r answer </dev/tty

    case "$answer" in
        y|Y)
            open -a "Google Chrome" "$url"
            ;;
        q|Q)
            echo "Stopped."
            break
            ;;
        *)
            ;;
    esac
  done < "$file"

    echo
    echo "Done."

    return
  fi

  echo "Title: \n" && read title
  echo "Link: \n" && read link

  echo "$title: $link" >> links.txt
}
