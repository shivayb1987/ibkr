req() {
  #!/bin/bash

# Check if the correct number of arguments is provided
if [ "$#" -lt 3 ]; then
  echo "Usage: $0 <Ticker> <Quantity> <Prices>"
  exit 1
fi

# Assign input values to variables
ticker=$1
quantity=$2
prices=$3

# Format the JSON output
output=$(cat <<EOF
{
  "orders": [
    {
      "prices": "$prices",
      "orderType": "STP LMT",
      "side": "BUY",
      "ticker": "$ticker",
      "tif": "GTC",
      "quantity": $(printf "%.0f" "$quantity")
    }
  ]
}
EOF
)

# Print the JSON output
echo "$output"
return "$output"

}

watch() {
  watchlist=$(grep WATCHLIST .watchlist)
  for symbol in "$@"
  do
    watchlist+=" $symbol"
  done

  echo $watchlist > .watchlist
}

unwatch() {
  #!/bin/bash

  watchlist_array=$(grep WATCHLIST .watchlist)

  delete_list_array=($@)

  for del_item in "${delete_list_array[@]}"; do
    watchlist_array=($(echo "${watchlist_array[@]}" | tr ' ' '\n' | grep -v "$del_item" | tr '\n' ' '))
  done

  WATCHLIST=$(echo "${watchlist_array[@]}")
  # echo "Modified WATCHLIST: $WATCHLIST"

  echo $WATCHLIST > .watchlist

}

pc() {
  perc=1
  num=$1

  if [ ! -z $2 ]; then
    perc=$2
  fi
  perc=$(bc -l <<< "scale=2; $perc/100")
  perc=$(($perc + 1))
  result=$(($num * $perc))
  result=$(printf "%.02f" "$result")
  echo $result
  echo $result | pbcopy
}
